```

import asyncio

from okx.websocket.WsPublicAsync import WsPublicAsync


def callbackFunc(message):
    print(message)


async def main():
    ws = WsPublicAsync(url="wss://wspap.okx.com:8443/ws/v5/business")
    await ws.start()
    args = [
        {
          "channel": "candle1m",
          "instId": "BTC-USDT"
        }
    ]

    await ws.subscribe(args, callback=callbackFunc)
    await asyncio.sleep(3600)

    # await ws.unsubscribe(args, callback=callbackFunc)
    # await asyncio.sleep(10)

asyncio.run(main())

```

```sql
-- 虛擬賬戶表
CREATE TABLE `tf_account` (
  `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '主鍵 ID',
  `account_no` VARCHAR(50) NOT NULL COMMENT '賬戶編號/策略標識',
  `initial_capital` DECIMAL(15, 2) NOT NULL COMMENT '初始資金',
  `available_cash` DECIMAL(15, 2) NOT NULL COMMENT '可用現金（可用於買入股票）',
  `frozen_cash` DECIMAL(15, 2) DEFAULT 0.00 COMMENT '凍結現金（掛單未成交或手續費預凍結）',
  `market_value` DECIMAL(15, 2) DEFAULT 0.00 COMMENT '持倉市值',
  `total_assets` DECIMAL(15, 2) NOT NULL COMMENT '總資產（可用現金 + 凍結現金 + 持倉市值）',
  `total_pnl` DECIMAL(15, 2) DEFAULT 0.00 COMMENT '累計盈虧金額',
  `total_pnl_pct` DECIMAL(8, 4) DEFAULT 0.0000 COMMENT '累計盈虧比例（收益率）',
  `status` TINYINT DEFAULT 1 COMMENT '狀態：1-正常，0-凍結',
  `create_time` DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '創建時間',
  `update_time` DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新時間',
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_account_no` (`account_no`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='回測虛擬賬戶表';

-- 資金流水明細表
CREATE TABLE `tf_fund_flow` (
  `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '主鍵 ID',
  `account_no` VARCHAR(50) NOT NULL COMMENT '賬戶編號',
  `trade_date` DATE NOT NULL COMMENT '業務發生日期',
  `biz_type` VARCHAR(30) NOT NULL COMMENT '業務類型：DEPOSIT-入金, WITHDRAW-出金, BUY-買入扣款, SELL-賣出回款, COMMISSION-傭金, TAX-印花稅',
  `amount` DECIMAL(15, 2) NOT NULL COMMENT '變動金額（收入為正，支出為負）',
  `balance_after` DECIMAL(15, 2) NOT NULL COMMENT '變動後的可用現金餘額',
  `related_order_id` VARCHAR(50) DEFAULT NULL COMMENT '關聯的訂單編號/流水號',
  `remark` VARCHAR(255) DEFAULT NULL COMMENT '備註說明',
  `create_time` DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '記錄時間',
  PRIMARY KEY (`id`),
  KEY `idx_account_date` (`account_no`, `trade_date`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='回測資金流水明細表';

-- 當前持倉表
CREATE TABLE `tf_position` (
  `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '主鍵 ID',
  `account_no` VARCHAR(50) NOT NULL COMMENT '賬戶編號',
  `symbol` VARCHAR(20) NOT NULL COMMENT '股票代碼（例如 002792.SZ）',
  `stock_name` VARCHAR(50) NOT NULL COMMENT '股票名稱',
  `total_volume` INT NOT NULL COMMENT '總持倉股數',
  `available_volume` INT NOT NULL COMMENT '可用股數（T+1：當天買入的為 0，次日解凍變為總股數）',
  `cost_price` DECIMAL(10, 3) NOT NULL COMMENT '持倉成本價（加權平均成本）',
  `current_price` DECIMAL(10, 3) NOT NULL COMMENT '當前最新價/收盤價',
  `market_value` DECIMAL(15, 2) NOT NULL COMMENT '當前持倉市值（total_volume * current_price）',
  `unrealized_pnl` DECIMAL(15, 2) DEFAULT 0.00 COMMENT '浮動盈虧金額',
  `unrealized_pnl_pct` DECIMAL(8, 4) DEFAULT 0.0000 COMMENT '浮動盈虧比例',
  `open_date` DATE NOT NULL COMMENT '建倉日期',
  `create_time` DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '創建時間',
  `update_time` DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新時間',
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_account_symbol` (`account_no`, `symbol`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='回測當前持倉表';

-- 歷史成交/持倉記錄表
CREATE TABLE `tf_trade_execution` (
  `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '主鍵 ID',
  `account_no` VARCHAR(50) NOT NULL COMMENT '賬戶編號',
  `order_id` VARCHAR(50) NOT NULL COMMENT '訂單編號',
  `symbol` VARCHAR(20) NOT NULL COMMENT '股票代碼',
  `stock_name` VARCHAR(50) NOT NULL COMMENT '股票名稱',
  `trade_side` VARCHAR(10) NOT NULL COMMENT '交易方向：BUY-買入, SELL-賣出',
  `trade_price` DECIMAL(10, 3) NOT NULL COMMENT '成交價格',
  `trade_volume` INT NOT NULL COMMENT '成交股數',
  `trade_amount` DECIMAL(15, 2) NOT NULL COMMENT '成交總金額（價格 * 股數）',
  `commission` DECIMAL(10, 2) DEFAULT 0.00 COMMENT '交易傭金',
  `tax` DECIMAL(10, 2) DEFAULT 0.00 COMMENT '印花稅（僅賣出時收取）',
  `realized_pnl` DECIMAL(15, 2) DEFAULT 0.00 COMMENT '平倉實現盈虧（僅賣出時計算）',
  `trade_time` DATETIME NOT NULL COMMENT '成交時間（回測中的模擬撮合時間點）',
  `create_time` DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '記錄時間',
  PRIMARY KEY (`id`),
  KEY `idx_account_symbol` (`account_no`, `symbol`),
  KEY `idx_trade_time` (`trade_time`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='回測歷史成交交割單表';
```
